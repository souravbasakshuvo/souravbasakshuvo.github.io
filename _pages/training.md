<script>
const state = {};
function goTo(id, total, idx) {
  state[id] = idx;
  document.getElementById(id).style.transform = `translateX(-${idx*100}%)`;
  document.querySelectorAll(`#${id}-dots span`).forEach((d,i) => d.style.opacity = i===idx?'1':'0.4');
}
function slide(id, total, dir) {
  const cur = (((state[id]||0) + dir) % total + total) % total;
  goTo(id, total, cur);
}
function autoSlide(id, total, ms) {
  setInterval(() => slide(id, total, 1), ms);
}
window.addEventListener('load', () => {
  state['s1'] = 0; state['s2'] = 0;
  autoSlide('s1', 4, 3000);
  autoSlide('s2', 3, 3000);
});
</script>
